```csharp
// Type(검사 대상) 별 z_index 상한. Phase 73 에서 지그마다 z 가 0 부터 독립 시작하도록 바뀌었다.
//  이 표는 레시피(FAI_1)에 종속적이다 — z 배치가 바뀌면 여기도 같이 고쳐야 한다.
//  -1 을 돌려주면 "이 Type 은 자동반복 대상이 아님" 이라는 뜻이다.
private static int ResolveMaxZIndexByType(int nTestType)
{
    switch (nTestType)
    {
        case 0: return 0;   // TOP    (z=0 단일)
        case 1: return 0;   // BOTTOM (z=0 단일)
        case 2: return 2;   // SIDE_1 (Datum 0,1 / 측정 2)
        case 3: return 3;   // SIDE_2 (Datum 0,1 / 측정 2,3)
        case 4: return 4;   // SIDE_3 (Datum 0,1 / 측정 2,3,4)
        case 5: return 3;   // SIDE_4 (Datum 0,1 / 측정 2,3)
        default: return -1;
    }
}
```

// 지그별 z 배치를 사람이 읽는 구간 이름으로 옮긴다. Phase 73 이후 z 는 지그마다 0 부터 시작한다.
//  표시 전용이라 틀려도 검사 결과에는 영향이 없지만, 로그 오독을 막기 위해 레시피와 맞춰 둔다.
private static string DescribeSideStage(int nTestType, int nZIndex)
