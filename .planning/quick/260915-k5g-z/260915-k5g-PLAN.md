using System;
using System.IO;
using System.Reflection;
using System.Runtime.CompilerServices;
using System.Text;

public static class ReviewerLabelProbe
{
    private static string s_binDir;

    public static int Main(string[] args)
    {
        if (args.Length < 3)
        {
            Console.Error.WriteLine("usage: ReviewerLabelProbe <binDir> <out.tsv> <dateDir>...");
            return 2;
        }
        s_binDir = args[0];
        AppDomain.CurrentDomain.AssemblyResolve += OnResolve;
        return Run(args);
    }

    private static Assembly OnResolve(object sender, ResolveEventArgs e)
    {
        string name = new AssemblyName(e.Name).Name;
        string dll = Path.Combine(s_binDir, name + ".dll");
        if (File.Exists(dll))
        {
            return Assembly.LoadFrom(dll);
        }
        string exe = Path.Combine(s_binDir, name + ".exe");
        if (File.Exists(exe))
        {
            return Assembly.LoadFrom(exe);
        }
        return null;
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    private static int Run(string[] args)
    {
        MethodInfo miInter = typeof(ReringProject.UI.ReviewerListLabelBuilder).GetMethod(
            "IsIntermediateTick", BindingFlags.Public | BindingFlags.Static);
        StringBuilder sb = new StringBuilder();
        int n = 0;
        for (int a = 2; a < args.Length; a++)
        {
            string dateDir = args[a];
            string dateName = Path.GetFileName(dateDir.TrimEnd('\\', '/'));
            string[] dirs = Directory.GetDirectories(dateDir);
            Array.Sort(dirs, StringComparer.Ordinal);
            foreach (string d in dirs)
            {
                string json = Path.Combine(d, "cycle.json");
                if (!File.Exists(json))
                {
                    continue;
                }
                ReringProject.UI.CycleResultDto dto = ReringProject.Sequence.CycleResultSerializer.Load(json);
                string label = ReringProject.UI.ReviewerListLabelBuilder.Build(dto);
                bool fail = ReringProject.UI.ReviewerListLabelBuilder.IsFailTick(dto);
                string inter = "-";
                if (miInter != null)
                {
                    object r = miInter.Invoke(null, new object[] { dto });
                    if ((bool)r) { inter = "1"; } else { inter = "0"; }
                }
                int z = -99;
                if (dto != null) { z = dto.ZIndex; }
                string szFail = "0";
                if (fail) { szFail = "1"; }
                sb.Append(dateName).Append('/').Append(Path.GetFileName(d)).Append('\t')
                  .Append(z).Append('\t').Append(szFail).Append('\t').Append(inter).Append('\t')
                  .Append(label).Append('\n');
                n++;
            }
        }
        File.WriteAllText(args[1], sb.ToString(), new UTF8Encoding(false));
        Console.WriteLine("rows=" + n);
        return 0;
    }
}
